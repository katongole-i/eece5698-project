cmake_minimum_required(VERSION 3.20)
project(ftran_u LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)

foreach(option BUILD_SHARED_LIBS BUILD_SHARED_EXTRAS_LIB BUILD_CXX_EXE
    BUILD_TESTING BUILD_EXAMPLES HIGHSINT64 HIPO ZLIB)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(FAST_BUILD ON CACHE BOOL "" FORCE)
add_subdirectory(HiGHS)
set_target_properties(highs highs_extras PROPERTIES UNITY_BUILD OFF)
target_compile_options(highs PRIVATE -ffp-contract=off)

option(FTRAN_U_XRT "Build the FPGA-enabled HiGHS executable" OFF)
if(FTRAN_U_XRT)
  find_package(XRT REQUIRED HINTS /opt/xilinx/xrt)
  add_executable(highs_fpga HiGHS/app/RunHighs.cpp HiGHS/highs/util/HFactor.cpp ftran_u_xrt.cpp)
  target_compile_features(highs_fpga PRIVATE cxx_std_17)
  target_compile_definitions(highs_fpga PRIVATE FTRAN_U_XRT)
  target_compile_options(highs_fpga PRIVATE -ffp-contract=off)
  target_include_directories(highs_fpga PRIVATE ${XRT_INCLUDE_DIRS})
  target_link_libraries(highs_fpga PRIVATE highs XRT::xrt_coreutil)
  return()
endif()

set(HLS_ROOT "/proj/octfpga-PG0/ikato/proj/ftran_u_top")
set(HLS_LDFLAGS "${HLS_ROOT}/build/lib/libhighs.a ${HLS_ROOT}/build/lib/libhighs_extras.a -pthread -ldl")
configure_file(hls_config.cfg.in "${CMAKE_CURRENT_SOURCE_DIR}/hls_config.cfg" @ONLY)
